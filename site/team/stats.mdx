# Stats

export const allMoments = [
  ...brockMoments,
  ...mistyMoments,
  ...surgeMoments,
  ...erikaMoments,
  ...sabrinaMoments,
  ...kogaMoments,
];

export const battleMoments = allMoments.filter((m) => m.kind === "battle");

export function StatsMoment() {
  const { value: storedLabel, set: setStoredLabel } = useStorageState("live-moment");
  const currentLabel = storedLabel ?? LIVE_MOMENT_DEFAULT;
  return (
    <Card title="Stats Setup">
      <Row
        row={[
          "Moment:",
          {
            dropdown: {
              value: currentLabel,
              options: battleMoments.map((m) => m.label),
              onChange: (v) => {
                if (v === LIVE_MOMENT_DEFAULT) removeState("live-moment");
                else setStoredLabel(v);
              },
            },
          },
        ]}
      />
    </Card>
  );
}

<StatsMoment />
